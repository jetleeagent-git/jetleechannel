# jetleechannel.sg - Static Site
# Old WordPress .htaccess backed up to .htaccess.wp.backup

# Serve index.html first (home.htm is corrupted)
DirectoryIndex index.html home.htm

# Disable HTML caching completely (bypass CDN issues)
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/jpg "access plus 1 year"
    ExpiresByType image/jpeg "access plus 1 year"
    ExpiresByType image/gif "access plus 1 year"
    ExpiresByType image/png "access plus 1 year"
    ExpiresByType image/webp "access plus 1 year"
    ExpiresByType text/css "access plus 1 month"
    ExpiresByType text/javascript "access plus 1 month"
    ExpiresByType text/html "access plus 0 seconds"
    ExpiresDefault "access plus 0 seconds"
</IfModule>

<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css text/javascript application/javascript application/json
</IfModule>

<IfModule mod_headers.c>
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "DENY"
    Header set Cache-Control "no-cache, must-revalidate" "expr=%{CONTENT_TYPE} =~ m|text/html|"
</IfModule>

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # Case-insensitive redirect for OneMarinaGardens (lowercase → correct case)
    RewriteRule ^onemarinagardens(/.*)?$ /OneMarinaGardens$1 [R=301,L,NC]

    # Don't apply the catch-all to project subdirectory files or directories
    RewriteRule ^(lentor-gardens|amberwood|hudsonplace|unionsquare|hougangcentral|lucernegrand|articles|blog|projects|thesen|elta|narraprices|narraresidences|aurea|Aurea|robertsonOpus|eltasingapore|dunearnhouse|thecontinuum|arcady|OneMarinaGardens|generations-tannery)(/.*)?$ - [L]

    # Blog redirect
    RewriteRule ^blog/?$ /blog/index.html [L]

    # Blog rewrite rules
    RewriteRule ^blog/([^/]+)/?$ /blog/$1/index.html [L]

    # If the request is not an existing file, serve index.html (SPA-like for direct paths)
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^(.*)$ index.html [L]

    # Rewrite home.htm to index.html (catch attempts to access it)
    RewriteRule ^home\.htm$ /index.html [L,R=301]
</IfModule>
